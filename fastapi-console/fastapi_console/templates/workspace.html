const CODE = "{{ unique_code }}";
let STATE = { containerStatus: 'stopped', addresses: [], hintViewed: false, completed: false, correctCount: 0, totalFlags: 0, score: 0, busy: false };
let SUBMISSIONS = [];
let stateHintShown = false;
let REFRESH_TIMER = null;

const logBox = document.getElementById('run-log');
const statusBadge = document.getElementById('status-badge');
const statusNote = document.getElementById('status-note');

// 本页原先**用了 esc() 却从没定义过它**（agent.html / dashboard.html 各有一份）。
// 一旦有提交记录，renderHistory() 就会 ReferenceError。补上；观察者面板也用它。
function esc(s) {
  const div = document.createElement('div');
  div.textContent = s == null ? '' : String(s);
  return div.innerHTML;
}

function appendLog(log) {
  const empty = logBox.querySelector('.dim');
  if (empty) empty.remove();
  // 计划内容来自 LLM，永远按纯文本显示；不能让不可信输出成为 HTML。
  logBox.insertAdjacentHTML('beforeend', logLine(esc(log.time), esc(log.type), esc(log.text)));
  logBox.scrollTop = logBox.scrollHeight;
}

function setBusy(busy, label) {
  STATE.busy = busy;
  document.getElementById('busy-state').textContent = busy ? (label || 'PROCESSING...') : '';
  ['btn-start', 'btn-hint', 'btn-round', 'btn-close', 'btn-submit'].forEach(id => {
    const el = document.getElementById(id);
    if (el) el.disabled = busy;
  });
}

function statusMeta() {
  const labels = {
    available: { label: '已就绪', tone: 'ready', note: '目标地址可通过靶场 VPN 访问。' },
    pending: { label: '启动中', tone: 'warming', note: '实例正在准备，地址出现后即可访问。' },
    stop_pending: { label: '停止中', tone: 'warming', note: '正在释放实例资源。' },
    stopped: { label: '已停止', tone: 'quiet', note: '启动容器后显示新的目标地址。' },
  };
  return labels[STATE.containerStatus] || { label: '未知状态', tone: 'quiet', note: '等待平台返回容器状态。' };
}

function renderState() {
  const meta = statusMeta();
  statusBadge.className = 'badge ' + meta.tone;
  statusBadge.innerHTML = '' + meta.label;
  statusNote.textContent = meta.note;
  const links = document.getElementById('target-links');
  links.innerHTML = STATE.addresses.length
    ? STATE.addresses.map(a => '' + a + ' ↗').join('')
    : '启动容器后显示入口地址';
  const pct = STATE.totalFlags ? Math.round((STATE.correctCount / STATE.totalFlags) * 100) : 0;
  document.getElementById('progress-bar').style.width = pct + '%';
  document.getElementById('progress-pct').textContent = pct + '%';
  document.getElementById('progress-detail').textContent = STATE.correctCount + ' / ' + STATE.totalFlags + ' captured';
  document.getElementById('score').textContent = STATE.score;
  document.getElementById('flag-count').textContent = STATE.correctCount + ' / ' + STATE.totalFlags + ' FLAGS';
  document.getElementById('btn-start').disabled = STATE.containerStatus === 'available' || STATE.busy;
  document.getElementById('btn-close').disabled = STATE.containerStatus !== 'available' || STATE.busy;
  document.getElementById('btn-hint').disabled = STATE.completed || STATE.hintViewed || STATE.busy;
  document.getElementById('btn-round').disabled = STATE.busy || STATE.completed;
  if (STATE.hintViewed) {
    const hintEl = document.getElementById('hint-panel');
    hintEl.style.display = 'flex';
    document.getElementById('hint-text').textContent = STATE.hintText || '该题没有可用提示。';
    document.getElementById('btn-hint').textContent = '已查看提示';
  }
}

function renderHistory() {
  const body = document.getElementById('history-body');
  document.getElementById('history-panel').style.display = SUBMISSIONS.length ? 'block' : 'none';
  document.getElementById('history-count').textContent = SUBMISSIONS.length + ' attempts';
  body.innerHTML = SUBMISSIONS.map((s, i) =>
    '' + String(i + 1).padStart(2, '0') + ''
    + '' + esc(s.flag) + ''
    + '' + (s.correct ? '√ 正确' : '× 错误') + ''
    + '' + (s.correct ? '+' + s.awarded : '—') + ''
  ).join('');
}

async function refresh() {
  const res = await fetch('/api/v1/challenges');
  const data = await res.json();
  const c = data.find(x => x.unique_code === CODE);
  if (!c) {
    if (!stateHintShown) {
      stateHintShown = true;
      // 查明模式并给出明确提示，避免无限刷屏
      let mode = '本地平台';
      try {
        const st = await fetch('/api/v1/state').then(r => r.json());
        mode = st.remote ? '远端平台（' + st.total_challenges + ' 题）' : '本地平台（' + st.total_challenges + ' 题）';
      } catch (e) { /* ignore */ }
      appendLog({ time: now(), type: 'error', text: '未找到题目 ' + CODE + ' —— 当前为' + mode + '，题库里没有该题' });
      appendLog({ time: now(), type: 'info', text: '如果该题属于远端任务，请到「设置」页填写 BENCHMARK_BASE_URL + BENCHMARK_TOKEN 并保存，然后刷新本页' });
      clearInterval(REFRESH_TIMER);
    }
    return;
  }
  stateHintShown = false;
  STATE.containerStatus = c.container_status;
  STATE.addresses = c.container_addr || [];
  STATE.correctCount = c.correct_flag_count;
  STATE.totalFlags = c.flag_count;
  STATE.completed = c.is_completed;
  document.getElementById('run-description').textContent = c.description || '暂无描述，使用目标地址与运行日志建立解题上下文。';
  const d = String(c.difficulty || '').toLowerCase();
  document.getElementById('level-pill').textContent = ({ easy: 'EASY', medium: 'MEDIUM', hard: 'HARD' }[d] || 'UNKNOWN') + ' · LV.' + c.level;
  renderState();
}

async function apiCall(path, method, payload) {
  const opts = { method: method || 'GET' };
  if (payload !== undefined) {
    opts.headers = { 'Content-Type': 'application/json', 'X-CSRFToken': 'csrf' };
    opts.body = JSON.stringify(payload);
  } else {
    opts.headers = { 'X-CSRFToken': 'csrf' };
  }
  const res = await fetch(path, opts);
  const data = await res.json();
  if (!res.ok) throw new Error('[' + data.code + '] ' + data.message);
  return data;
}

async function run(method, label) {
  try {
    await method();
  } catch (err) {
    appendLog({ time: now(), type: 'error', text: label + '失败 ' + err.message });
  } finally {
    setBusy(false);
  }
}

  appendLog({ time: now(), type: 'api', text: '启动容器 ' + CODE + ' ...' });
  const res = await apiCall('/api/v1/challenges/start?unique_code=' + encodeURIComponent(CODE), 'POST');
  STATE.containerStatus = 'available';
  STATE.addresses = res.container_addr || [];
  appendLog({ time: now(), type: 'success', text: '容器已就绪: ' + STATE.addresses.join(', ') });
  renderState();
}, '启动'));

document.getElementById('btn-hint').addEventListener('click', () => run(async () => {
  setBusy(true, 'FETCHING...');
  const res = await apiCall('/api/v1/challenges/hint?unique_code=' + encodeURIComponent(CODE), 'GET');
  STATE.hintViewed = true;
  STATE.hintText = res.hint;
  appendLog({ time: now(), type: 'info', text: '已获取提示（后续提交将按比例扣分）: ' + (res.hint || '(无提示)') });

}, '获取提示'));

document.getElementById('btn-round').addEventListener('click', () => run(async () => {

  const res = await apiCall('/api/v1/ai/round', 'POST', { unique_code: CODE });
  (res.logs || []).forEach(appendLog);
  if (res.plan) appendLog({ time: now(), type: 'info', text: '解题计划:\n' + res.plan });
  await refresh();
}, 'AI 计划'));

document.getElementById('btn-close').addEventListener('click', () => run(async () => {
  setBusy(true, 'CLOSING...');
  const res = await apiCall('/api/v1/challenges/close?unique_code=' + encodeURIComponent(CODE), 'POST');
  STATE.containerStatus = 'stopped';
  STATE.addresses = [];
  appendLog({ time: now(), type: 'success', text: '容器已关闭，资源已释放' });
  renderState();
}, '关闭'));

document.getElementById('btn-submit').addEventListener('click', async () => {
  const input = document.getElementById('manual-flag');
  const flag = input.value.trim();
  if (!flag) return;
  try {
    const res = await apiCall('/api/v1/challenges/submit', 'POST', { unique_code: CODE, flag });
    SUBMISSIONS.push({ flag, correct: res.correct, awarded: res.awarded });
    if (res.correct) {
      const before = Number(STATE.correctCount || 0);
      const reported = Number(res.correct_flag_count || 0);
      STATE.score = res.cumulative_score;
      const expected = Math.max(1, Number(STATE.totalFlags || 1));
      const reportedTotal = Number(res.total_flag_count || 0);
      const total = Math.max(expected, reportedTotal);
      if (reportedTotal > 0) {
        STATE.correctCount = Math.max(before, reported);
      } else if (expected === 1) {
        STATE.correctCount = Math.max(before, 1);
      } else {
        STATE.correctCount = Math.min(expected, Math.max(before, reported) + 1);
      }
      const complete = reportedTotal > 0
        ? reported >= total
        : expected === 1 ? !!res.correct : STATE.correctCount >= expected;
      if (complete) STATE.completed = true;
      appendLog({ time: now(), type: 'success', text: '√ 正确: ' + flag + ' (+' + res.awarded + ') 累计 ' + res.cumulative_score });
    } else {
      appendLog({ time: now(), type: 'error', text: '× 错误: ' + flag });
    }
    renderHistory();
    renderState();
    input.value = '';
  } catch (err) {
    appendLog({ time: now(), type: 'error', text: '提交失败 ' + err.message });
  }
});

document.getElementById('manual-flag').addEventListener('keyup', e => {
  if (e.key === 'Enter') document.getElementById('btn-submit').click();
});

// ── 观察者之镜（B61）───────────────────────────────────────
// 只读展示 work//.heimdall.json。服务端用 adapter.heimdall.render() 渲染，
// 所以这里显示的就是主 Agent 当时收到的**同一张图**（打码处会显式标注）。
// 抽成具名函数是为了能被 Node 自测直接调用（B60 的教训：埋在大函数里的逻辑测不到）。
const MIRROR = { timer: null };

function mirrorChip(label, n) {
  return '' + label + '' + (n || 0) + '';
}

function renderMirror(d) {
  const state = document.getElementById('mirror-state');
  const counts = document.getElementById('mirror-counts');
  const body = document.getElementById('mirror-body');
  const notes = document.getElementById('mirror-notes');
  const gone = document.getElementById('mirror-gone');
  const meta = document.getElementById('mirror-meta');

  counts.style.display = 'none';
  counts.innerHTML = '';
  body.style.display = 'none';
  body.textContent = '';        // 永远走 textContent：那是 LLM 的输出，不能当 HTML
  notes.innerHTML = '';
  gone.innerHTML = '';

  if (!d || !d.available) {
    meta.textContent = '—';
    state.className = 'mirror-state';
    let why = (d && d.reason) || '观察者还没有为这道题出过图';
    if (d && d.updated_text) why += '（最后一次观察 ' + d.updated_text + '）';
    state.textContent = why;
    return;
  }

  state.className = 'mirror-state dim';
  state.textContent = '这是另一路 Agent 读主 Agent 的思考画出的镜子，注入下一场的 prompt；'
    + '它只举镜子、不发指令，且每条都可能错。';
  meta.textContent = '第 ' + d.session + ' 场' + (d.updated_text ? ' · ' + d.updated_text : '');

  counts.innerHTML = mirrorChip('LOCK', d.counts.lock) + mirrorChip('DEAD', d.counts.dead)
    + mirrorChip('ANGLES', d.counts.angles) + mirrorChip('POSTPONE', d.counts.postpone)
    + mirrorChip('TENSION', d.tension_total);
  counts.style.display = 'flex';

  if (d.map_text) { body.textContent = d.map_text; body.style.display = 'block'; }

  const n = [];
  if (d.redacted > 0) {
    n.push('⚠ 已按合规口径打码 ' + d.redacted
      + ' 处 —— 这不是主 Agent 原样看到的字面');
  }
  const drops = Object.entries(d.dropped_by_cap || {});
  const dtot = drops.reduce(function (a, kv) { return a + kv[1]; }, 0);
  if (dtot > 0) {
    n.push('另有 ' + dtot + ' 条超出每类显示上限，未进入注入正文（'
      + drops.map(function (kv) { return kv[0] + ' ' + kv[1]; }).join('、') + '）');
  }
  if (d.char_trimmed) {
    n.push('注入正文另因字数预算被截断过（镜像不该抢占注意力）');
  }
  notes.innerHTML = n.join('');

  // 撤销与淡出措辞必须分开：撤销 = "这条错了"，淡出 = "这条没再被观察到"。
  // 混在一起会让人把后者读成前者 —— 而误判死路正是观察者红线 2 要防的事。
  let g = '';
  if (d.retracted && d.retracted.length) {
    g += '
曾标出但已撤销'
      + d.retracted.map(function (x) {
          return '
' + esc(x.id) + ' '
            + esc(x.text) + (x.why ? ' —— ' + esc(x.why) + '' : '') + '
';
        }).join('')
      + '
';
  }
  if (d.faded && d.faded.length) {
    g += '
已淡出（不一定是错的）'
      + d.faded.map(function (x) {
          return '
' + esc(x.id) + ' '
            + esc(x.text) + '
';
        }).join('')
      + '
';
  }
  gone.innerHTML = g;
}

async function refreshMirror() {
  try {
    const res = await fetch('/api/v1/challenges/heimdall?unique_code=' + encodeURIComponent(CODE));
    renderMirror(await res.json());
  } catch (e) {
    renderMirror({ available: false, reason: '观察者接口不可达：' + e.message });
  }
}
